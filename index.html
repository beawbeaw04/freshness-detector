function analyze() { if (video.readyState === video.HAVE_ENOUGH_DATA) { canvas.width = 64; canvas.height = 64; ctx.drawImage(video, video.videoWidth/2, video.videoHeight/2, 1, 1, 0, 0, 1, 1); const [r, g, b] = ctx.getImageData(0, 0, 1, 1).data; const hex = `#${toHex(r)}${toHex(g)}${toHex(b)}`; colorDot.style.backgroundColor = hex; hexLabel.innerText = hex; rgbDebug.innerText = `RGB: (${r}, ${g}, ${b})`; const brightness = (r + g + b) / 3; let freshness = 0; let colorName = ""; if (brightness > 30) { // ปรับปรุงลอจิกใหม่: ใช้สัดส่วนและการเปรียบเทียบค่าสีเพื่อแก้ปัญหาการอ่านสีม่วงไม่ได้ if (g <= 65 && r >= 90) { // กลุ่ม 1: ม่วงเข้ม (110, 40, 150) -> ค่า G จะต่ำมากและ R จะสูงชัดเจน colorName = "💜 สีม่วงเข้ม"; freshness = Math.round(100 - ((g - 40) / 25) * 10); } else if (g <= 95 && b >= 140 && (b - g) >= 50) { // กลุ่ม 2: ม่วงอมฟ้า (85, 80, 175) -> ปลดล็อกเงื่อนไข R ออก แล้วเช็คว่า B ต้องมากกว่า G อย่างน้อย 50 แต้ม colorName = "💙 สีม่วงอมฟ้า"; freshness = Math.round(89 - ((g - 70) / 25) * 19); } else if (g >= 85 && g <= 120 && b >= 140 && (b - g) < 50) { // กลุ่ม 3: น้ำเงิน (55, 100, 190) -> ค่า G จะพุ่งขึ้นมาใกล้เคียงกับ B มากขึ้น (ห่างกันไม่ถึง 50 แต้ม) colorName = "🔷 สีน้ำเงิน"; freshness = Math.round(69 - ((g - 90) / 25) * 29); } else { // กลุ่ม 4: เขียวอมฟ้า (40, 150, 170) -> ค่า G สูงโดดเด่นและแซงหน้าหรือใกล้เคียง B มาก colorName = "🟢 สีเขียวอมฟ้า"; freshness = Math.round(39 - ((g - 110) / 50) * 39); } if (freshness > 100) freshness = 100; if (freshness < 0) freshness = 0; } scoreNum.innerText = freshness; scoreFill.style.width = freshness + "%"; // การแสดงผลแยกตามช่วงระดับความสดจริง if (freshness >= 70) { updateUI(`✅ สดมาก\n[${colorName}]`, "สีฟิล์มปกติ อาหารมีความสดสูง บริโภคได้อย่างปลอดภัย", "#28a745"); } else if (freshness >= 40) { updateUI(`⚠️ เริ่มเปลี่ยนแปลง\n[${colorName}]`, "เริ่มมีการปล่อยก๊าซ/เสื่อมสภาพ ควรระมัดระวังในการใช้งาน", "#ffc107"); } else if (brightness <= 30) { updateUI("🔍 ค้นหา...", "ขยับกล้องส่องแผ่นฟิล์มวัดสีให้ตรงจุดกึ่งกลาง", "#888"); scoreFill.style.width = "0%"; } else { updateUI(`❌ อันตราย!\n[${colorName}]`, "ด่างหรือก๊าซเน่าเสียสูงเกินเกณฑ์มาตรฐาน ห้ามบริโภคเด็ดขาด", "#dc3545"); } } requestAnimationFrame(analyze); }
