function analyze() { if (video.readyState === video.HAVE_ENOUGH_DATA) { canvas.width = 64; canvas.height = 64; ctx.drawImage(video, video.videoWidth/2, video.videoHeight/2, 1, 1, 0, 0, 1, 1); const [r, g, b] = ctx.getImageData(0, 0, 1, 1).data; const hex = `#${toHex(r)}${toHex(g)}${toHex(b)}`; colorDot.style.backgroundColor = hex; hexLabel.innerText = hex; rgbDebug.innerText = `RGB: (${r}, ${g}, ${b})`; const brightness = (r + g + b) / 3; let freshness = 0; let colorName = ""; if (brightness <= 30) { // หากมืดเกินไป ให้ระบบรอการสแกน updateUI("🔍 ค้นหา...", "ขยับกล้องส่องแผ่นฟิล์มวัดสีให้ตรงจุดกึ่งกลาง", "#888"); scoreNum.innerText = 0; scoreFill.style.width = "0%"; } else { // ล็อกเงื่อนไขตามช่วงค่าสี RGB จริงที่คุณระบุมา if (g <= 60 && b >= 130) { // 1. กลุ่มสีม่วงเข้ม (ค่า G ต่ำมาก อยู่ในช่วง 40-50) -> สดมาก 100% colorName = "💜 สีม่วงเข้ม"; freshness = 100; updateUI(`✅ สดมาก [${colorName}]`, "สีฟิล์มปกติ อาหารมีความสดสูง บริโภคได้อย่างปลอดภัย", "#28a745"); } else if (g > 60 && g <= 92 && b >= 165) { // 2. กลุ่มสีม่วงอมฟ้า (ค่า G ขยับขึ้นมาช่วง 80-90 และ B สูง) -> สดมาก 85% colorName = "💙 สีม่วงอมฟ้า"; freshness = 85; updateUI(`✅ สดมาก [${colorName}]`, "สีฟิล์มปกติ อาหารมีความสดสูง บริโภคได้อย่างปลอดภัย", "#28a745"); } else if (g > 90 && g <= 110 && b >= 170) { // 3. กลุ่มสีน้ำเงิน (ค่า G เริ่มสูงแตะ 100) -> เริ่มเปลี่ยนแปลง 50% colorName = "🔷 สีน้ำเงิน"; freshness = 50; updateUI(`⚠️ เริ่มเปลี่ยนแปลง [${colorName}]`, "เริ่มมีการปล่อยก๊าซ/เสื่อมสภาพ ควรระมัดระวังในการใช้งาน", "#ffc107"); } else { // 4. กลุ่มสีเขียวอมฟ้า หรือสีอื่น ๆ (ค่า G พุ่งสูงเกิน 110 ขึ้นไป) -> อันตราย 0% colorName = "🟢 สีเขียวอมฟ้า"; freshness = 0; updateUI(`❌ อันตราย! [${colorName}]`, "ด่างหรือก๊าซเน่าเสียสูงเกินเกณฑ์มาตรฐาน ห้ามบริโภคเด็ดขาด", "#dc3545"); } // อัปเดตตัวเลขและแถบเปอร์เซ็นต์บนหน้าจอ scoreNum.innerText = freshness; scoreFill.style.width = freshness + "%"; scoreFill.style.backgroundColor = (freshness >= 70) ? "#28a745" : (freshness >= 40 ? "#ffc107" : "#dc3545"); } } requestAnimationFrame(analyze); }
