function analyze() { if (video.readyState === video.HAVE_ENOUGH_DATA) { canvas.width = 64; canvas.height = 64; ctx.drawImage(video, video.videoWidth/2, video.videoHeight/2, 1, 1, 0, 0, 1, 1); const [r, g, b] = ctx.getImageData(0, 0, 1, 1).data; const hex = `#${toHex(r)}${toHex(g)}${toHex(b)}`; colorDot.style.backgroundColor = hex; hexLabel.innerText = hex; rgbDebug.innerText = `RGB: (${r}, ${g}, ${b})`; const brightness = (r + g + b) / 3; let freshness = 0; let colorName = ""; if (brightness <= 30) { updateUI("🔍 ค้นหา...", "ขยับกล้องส่องแผ่นฟิล์มวัดสีให้ตรงจุดกึ่งกลาง", "#888"); scoreNum.innerText = 0; scoreFill.style.width = "0%"; } else { // ลอจิกใหม่อิงตามสัดส่วนสีจริงจากหน้างาน (แก้ไขปัญหากล้องเจอแสงจ้าแล้วค่าพุ่ง) if (b > g && r > g && (b - g) >= 15) { // 1. กลุ่มสีม่วง (ม่วงเข้ม/ม่วงอมฟ้า): ค่า B และ R ต้องชนะ G เสมอ แม้แสงจะจ้ามาก (เช่นกรณี R:216 G:202 B:231) if (g < 100) { colorName = "💜 สีม่วงเข้ม"; freshness = 100; } else { colorName = "💙 สีม่วงอมฟ้า"; freshness = 85; } updateUI(`✅ สดมาก [${colorName}]`, "สีฟิล์มปกติ อาหารมีความสดสูง บริโภคได้อย่างปลอดภัย", "#28a745"); } else if (b > r && g > r && b > g) { // 2. กลุ่มสีน้ำเงิน: ค่า B จะสูงโดดเด่นแซงทั้ง G และ R colorName = "🔷 สีน้ำเงิน"; freshness = 50; updateUI(`⚠️ เริ่มเปลี่ยนแปลง [${colorName}]`, "เริ่มมีการปล่อยก๊าซ/เสื่อมสภาพ ควรระมัดระวังในการใช้งาน", "#ffc107"); } else { // 3. กลุ่มสีเขียวอมฟ้า: ค่า G จะพุ่งขึ้นมาสูงเด่นเทียบเท่าหรือแซงค่า B และค่า R จะต่ำที่สุด colorName = "🟢 สีเขียวอมฟ้า"; freshness = 0; updateUI(`❌ อันตราย! [${colorName}]`, "ด่างหรือก๊าซเน่าเสียสูงเกินเกณฑ์มาตรฐาน ห้ามบริโภคเด็ดขาด", "#dc3545"); } // อัปเดตตัวเลขและแถบเปอร์เซ็นต์บนหน้าจอ scoreNum.innerText = freshness; scoreFill.style.width = freshness + "%"; scoreFill.style.backgroundColor = (freshness >= 70) ? "#28a745" : (freshness >= 40 ? "#ffc107" : "#dc3545"); } } requestAnimationFrame(analyze); }
